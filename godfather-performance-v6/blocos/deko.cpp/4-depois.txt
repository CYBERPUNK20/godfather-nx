        if (!capture_start) capture_start = draw_tick;
        if (draw_tick - capture_start >= 80'000'000'000ull && captured < 1024) {
            unsigned id = captured++;
            uint32_t texture_mask = 0;
            for (unsigned i = 0; i < 4; ++i)
                if (s.textures[i]) texture_mask |= 1u << i;
            float px = 0, py = 0, pz = 0, pw = 0;

                    const float* p = &a.vertices[size_t(v) * std::popcount(a.regs) * 4];
                    px = p[0]; py = p[1]; pz = p[2]; pw = p[3];
                }

                    miny = std::min(miny, p[1]); maxy = std::max(maxy, p[1]);
