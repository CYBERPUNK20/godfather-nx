    if constexpr (logx::kVerbose) {
        const bool fvf_weights = !(s.vs_handle & 1) && (s.vs_handle & 0xE) >= 0x6 && (s.vs_handle & 0xE) <= 0xC;
        const bool decl_weights = (s.vs_handle & 1) && s.vs && s.vs->inputs[1].type != 0;
        if (s.rs[kRsVertexBlend] || fvf_weights || decl_weights) {
            static std::unordered_map<uint64_t, bool> reports;
            uint32_t stride = a.buffer_count ? a.buffers[0].stride : 0;
            uint32_t bytes = a.buffer_count ? a.buffers[0].bytes : 0;
            uint32_t input1 = (s.vs_handle & 1) && s.vs ? s.vs->inputs[1].type : 0;
            uint64_t report_key = Hash(&s.vs_handle, sizeof(s.vs_handle));
            report_key = Hash(&s.rs[kRsVertexBlend], sizeof(s.rs[kRsVertexBlend]), report_key);
            report_key = Hash(&a.regs, sizeof(a.regs), report_key);
            report_key = Hash(&a.decode, sizeof(a.decode), report_key);
            report_key = Hash(&stride, sizeof(stride), report_key);
            if (reports.size() < 256 && reports.emplace(report_key, true).second) {
                LOGR("blend trace: fvf/handle=%08X state=%08X program=%u weights=%u in1=%08X regs=%04X vertices=%u stride=%u bytes=%u indices=%u",
                     s.vs_handle, s.rs[kRsVertexBlend], (s.vs_handle & 1) && s.vs && !s.vs->program.empty(),
                     fvf_weights || decl_weights, input1, a.regs, a.vertex_count, stride, bytes, a.index_count);
            }
        }
    }
