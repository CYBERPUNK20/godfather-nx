    if (sh.prog && a.vertex_count >= 100) {
        static uint64_t dumped[4] = {};
        static size_t dumped_count = 0;
        bool seen = false;
        for (size_t i = 0; i < dumped_count; i++) seen |= dumped[i] == sh.vkey;
        if (!seen && dumped_count < std::size(dumped)) {
            dumped[dumped_count++] = sh.vkey;
            LOGR("VSDBG handle=%08X instructions=%zu regs=%04X decode=%016llX vertices=%u stride=%u hlsl=%zu",
                 s.vs_handle, s.vs->program.size() / 4, a.regs, static_cast<unsigned long long>(a.decode), a.vertex_count,
                 a.buffer_count ? a.buffers[0].stride : 0, src.size());
            std::string flat;
            flat.reserve(src.size());
            for (char c : src) flat.push_back(c == '\n' || c == '\r' ? ' ' : c);
            for (size_t off = 0; off < flat.size(); off += 360) {
                std::string chunk = flat.substr(off, 360);
                LOGR("VSDBG %04zu/%zu %s", off, flat.size(), chunk.c_str());
            }
        }
    }
