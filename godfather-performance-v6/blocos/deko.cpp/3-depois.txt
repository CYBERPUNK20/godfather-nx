        static bool dumped_scene_program = false;
        if (!dumped_scene_program) {
            dumped_scene_program = true;
            LOGR("SCENEVS handle=%08X instructions=%zu type=%08X regs=%04X decode=%016llX vertices=%u decl=%zu stride0=%u input0=(stream=%d offset=%u type=%02X) constants=%zu",

                 static_cast<unsigned long long>(a.decode), a.vertex_count, s.vs->decl.size(), s.vs->stream_stride[0],
                 s.vs->inputs[0].stream, s.vs->inputs[0].offset, s.vs->inputs[0].type, s.vs->constants.size());
            for (size_t n = 0; n < s.vs->decl.size() && n < 64; n++)
                LOGR("SCENEVS decltoken[%zu]=%08X", n, s.vs->decl[n]);
            for (const auto& [reg, v] : s.vs->constants)
                if (reg >= 90 && reg <= 100)
                    LOGR("SCENEVS declconst c%u=(%.6g %.6g %.6g %.6g)", reg, v[0], v[1], v[2], v[3]);
            for (size_t n = 0; n + 4 <= s.vs->program.size(); n += 4)
                LOGR("SCENEVS token[%zu]=%08X %08X %08X %08X", n / 4, s.vs->program[n], s.vs->program[n + 1],
                     s.vs->program[n + 2], s.vs->program[n + 3]);
            std::string src = VertexHlsl(s, a, sh), flat;
            flat.reserve(src.size());
            for (char c : src) flat.push_back(c == '\n' || c == '\r' ? ' ' : c);
            for (size_t off = 0; off < flat.size(); off += 360) {
                std::string chunk = flat.substr(off, 360);
                LOGR("SCENEVS hlsl %04zu/%zu %s", off, flat.size(), chunk.c_str());
            }
        }
